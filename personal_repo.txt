Manual Commands to Run All Services
1. Build all services

cd ~/Desktop/Project/pdev/auth-service && ~/sdk/go1.22.5/bin/go build -o bin/auth-service ./cmd
cd ~/Desktop/Project/pdev/jobs-service && ~/sdk/go1.22.5/bin/go build -o bin/jobs-service ./cmd
cd ~/Desktop/Project/pdev/api-gateway && ~/sdk/go1.22.5/bin/go build -o bin/api-gateway ./cmd
2. Start services (run each in a separate terminal)

# Terminal 1 — Auth Service (:8081)
cd ~/Desktop/Project/pdev/auth-service && ./bin/auth-service

# Terminal 2 — Jobs Service (:8082)
cd ~/Desktop/Project/pdev/jobs-service && ./bin/jobs-service

# Terminal 3 — API Gateway (:8080)
cd ~/Desktop/Project/pdev/api-gateway && ./bin/api-gateway

# Terminal 4 — Frontend (:5173)
cd ~/Desktop/Project/pdev/frontend && npm run dev
3. Or run all in background (single terminal)

cd ~/Desktop/Project/pdev/auth-service && nohup ./bin/auth-service > /tmp/auth.log 2>&1 &
cd ~/Desktop/Project/pdev/jobs-service && nohup ./bin/jobs-service > /tmp/jobs.log 2>&1 &
cd ~/Desktop/Project/pdev/api-gateway && nohup ./bin/api-gateway > /tmp/gateway.log 2>&1 &
cd ~/Desktop/Project/pdev/frontend && npm run dev
4. Stop all services

fuser -k 8080/tcp 8081/tcp 8082/tcp 5173/tcp
5. View logs

tail -f /tmp/auth.log      # auth service
tail -f /tmp/jobs.log       # jobs service
tail -f /tmp/gateway.log    # gateway
6. Or use the Makefile shortcuts

make build    # build all
make dev      # build + start all
make stop     # kill all
